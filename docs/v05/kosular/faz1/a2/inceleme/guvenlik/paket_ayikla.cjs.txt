const fs = require('fs');
const html = fs.readFileSync(process.argv[2], 'utf8');
const scripts = [...html.matchAll(/<script>([\s\S]*?)<\/script>/g)].map((m) => m[1]);
// oyun paketi: dk-test-kancalari-v1 işaretini içeren betik
const oyun = scripts.find((s) => s.includes('dk-test-kancalari-v1'));
fs.writeFileSync(process.argv[3], oyun);
console.log('betik sayısı', scripts.length, 'oyun paketi', oyun.length);

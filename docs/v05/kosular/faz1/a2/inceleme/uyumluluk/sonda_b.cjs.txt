const yerel = (u) => /^(https?|wss?):\/\/(127\.0\.0\.1|localhost)(:\d+)?\//.test(u) || /^(file|data|blob):/.test(u);

async function baglam(b, kayit) {
  const ctx = await b.newContext({ viewport: { width: 640, height: 360 } });
  kayit.disIstek = []; kayit.disWs = [];
  await ctx.route((u) => !yerel(u.href), (r) => { kayit.disIstek.push(r.request().url().slice(0, 200)); r.abort(); });
  await ctx.routeWebSocket((u) => !yerel(u.href), (ws) => { kayit.disWs.push(ws.url().slice(0, 200)); ws.close({ code: 1011, reason: 'inceleme: dış bağlantı kesildi' }); });
  return ctx;
}
async function sayfa(b, ad, url, { yolla = null, hazirBekle = true, hazirZaman = 90000 } = {}) {
  const kayit = { url, konsolHata: [], sayfaHata: [], istek4xx: [], ws: [] };
  const ctx = await baglam(b, kayit);
  if (yolla) await ctx.route('**/js/test/kancalar.js', yolla);
  const p = await ctx.newPage();
  p.on('console', (m) => { if (m.type() === 'error') kayit.konsolHata.push(m.text().slice(0, 300)); });
  p.on('pageerror', (e) => kayit.sayfaHata.push(String(e.message).slice(0, 300)));
  p.on('response', (r) => { if (r.status() >= 400) kayit.istek4xx.push(r.status() + ' ' + r.url().slice(0, 150)); });
  p.on('websocket', (w) => kayit.ws.push(w.url().slice(0, 200)));
  await p.addInitScript(() => { let h; window.__incHazir = null; Object.defineProperty(window, '__ready', { configurable: true, enumerable: true, get() { return h; }, set(v) { h = v; if (v && !window.__incHazir) window.__incHazir = { dkTipi: typeof window.DK, t: Math.round(performance.now()) }; } }); });
  const t0 = Date.now();
  await p.goto(url);
  if (hazirBekle) { try { await p.waitForFunction('window.__ready === true', null, { timeout: hazirZaman }); kayit.hazirMs = Date.now() - t0; } catch (e) { kayit.hazirHata = e.message.slice(0, 160); } }
  sonuc.sayfalar[ad] = kayit;
  return { ctx, p, kayit };
}

def satirlar(metin):
    out = {}
    for l in metin.splitlines():
        if not l.startswith('| KI-'): continue
        h = [x.strip() for x in l.strip().strip('|').split('|')]
        if len(h) != 8: continue                     # 'Kapanan maddeler' tablosu farklı sütunlu
        out[h[0]] = dict(zip(['kimlik', 'baslik', 'oncelik', 'alan', 'dosya', 'faz', 'durum', 'kaynak'], h))
    return out

DENETIM = re.compile(r'\b([A-K]-M?\d{2})\b')
def denetim_ayir(kaynak):
    birincil = kaynak.split(' / eski:', 1)[0]
    ana_k, ilgili = [], []
    for parca in birincil.split(';'):
        p = parca.strip()
        hedef = ilgili if p.startswith('ilgili:') else ana_k
        for m in DENETIM.findall(p):
            if m not in hedef: hedef.append(m)
    # 'İlgili: K-M01' biçimi 'eski:' kısmında da geçebilir (KI-233)
    for m in re.findall(r'[İi]lgili:\s*([A-K]-M?\d{2})', kaynak):
        if m not in ilgili: ilgili.append(m)
    return ana_k, [x for x in ilgili if x not in ana_k]

def ki_durumu(k, r):
    if k not in ana: return 'duzeltildi' if k in kapanan else None
    if r['durum'].startswith('Kısmen düzeltildi'): return 'kismi'
    if r['durum'].startswith('Açık'): return 'acik'
    return None
def faz_ayir(f):
    n = [int(x) for x in re.findall(r'Faz (\d)', f)]
    return (n[0] if n else None), n[1:]
sorunlar = {}

    k = f'KI-{i:03d}'
    r = ana.get(k) or ilk.get(k)
    if r is None: raise SystemExit(f'{k} bulunamadı')
    faz, ek = faz_ayir(r['faz'])
    den, ilg = denetim_ayir(r['kaynak'])
    notlar = []
    durum = ki_durumu(k, r)
    if durum is None: raise SystemExit(f"{k}: KNOWN_ISSUES durumu tanınmadı ({r['durum'][:60]})")
    kap = KAPANIS.get(k)
    if durum != 'acik' and (not kap or kap['durum'] != durum): raise SystemExit(f'{k}: KNOWN_ISSUES durumu {durum}, KAPANIS tablosunda karşılığı yok ya da farklı')
    if kap and kap['durum'] != durum: raise SystemExit(f"{k}: KAPANIS {kap['durum']} ↔ KNOWN_ISSUES {durum}")

        notlar.append(kap['not'])
    elif r['durum'] != 'Açık':

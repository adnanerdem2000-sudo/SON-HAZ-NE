def kip(ad, test):
    if len(KOSULAR[test]) == 1: return 'v0.4 taklit'
    m = re.match(r'^.*_(normal|noworker|crash|badurl)(?:_(\d))?$', ad)
    return m.group(1) + (f" #{m.group(2)}" if m.group(2) else '')
RAP = {}
def rapor(ad):
    if ad not in RAP:
        with open(os.path.join(Y, ad + '.json'), encoding='utf8') as f: RAP[ad] = json.load(f)
    return RAP[ad]
def kisa(v, n=60):
    s = v if isinstance(v, str) else json.dumps(v, ensure_ascii=False)
    return s if len(s) <= n else s[:n - 1] + '…'

eslenemeyen, sayac = [], {'YESIL': 0, 'KIRMIZI': 0, 'UYGULANAMAZ': 0}
for r in S:
    t, kid = r['test'], r['kimlik']
    if t.endswith('.mjs'):
        r['once_v04_taklit'] = r['once_v04']
        r['once_v04_taklit_aciklama'] = "Node testi: işaretçi kilidinden bağımsız, yinelenmedi; once_v04 ile aynı."
    elif r['once_v04'] == 'UYGULANAMAZ':
        r['once_v04_taklit'] = 'UYGULANAMAZ'
        r['once_v04_taklit_aciklama'] = "v0.4'te bu kip yok; taklitle de koşulmadı."
    else:
        parcalar, oklar = [], []
        for ad in KOSULAR[t]:
            K = rapor(ad)['kontroller']
            if kid.startswith('eski 12 kontrol'):
                sec = [k for k in K if not re.match(r'^[A-Z]{2,3}[-]?\d', k['id'])]
                ok = all(k['ok'] for k in sec)
                parcalar.append(f"{kip(ad, t)}: {'geçti' if ok else 'KALDI'} ({sum(k['ok'] for k in sec)}/{len(sec)})"); oklar.append(ok)
                continue
            k = next((k for k in K if k['id'] == kid), None)
            if k is None:
                parcalar.append(f"{kip(ad, t)}: kontrol yok"); continue
            parcalar.append(f"{kip(ad, t)}: {'geçti' if k['ok'] else 'KALDI'} ({kisa(k.get('deger'))})"); oklar.append(bool(k['ok']))
        if not oklar:
            eslenemeyen.append((t, kid)); r['once_v04_taklit'] = 'ESLENEMEDI'
        else:
            # B0'ın kuralı: bir kip (ya da tekrar) bile kalırsa KIRMIZI; karışık sonuç açıklamada görünür
            r['once_v04_taklit'] = 'YESIL' if all(oklar) else 'KIRMIZI'
        r['once_v04_taklit_aciklama'] = '; '.join(parcalar)
    if (t, kid) in NOTLAR: r['not_taklit'] = NOTLAR[(t, kid)]
    sayac[r['once_v04_taklit']] = sayac.get(r['once_v04_taklit'], 0) + 1

degisen = [(r['test'], r['kimlik'], r['once_v04'], r['once_v04_taklit']) for r in S if r['once_v04'] != r['once_v04_taklit']]

    f.write(json.dumps({'sema': 'dk-69-siniflandirma/1', 'kaynak': 'siniflandirma.json (B0, değişmedi) + once_v04_taklit/*.json (#69)',
                        'sayim': sayac, 'satirlar': cikti}, ensure_ascii=False, indent=1) + '\n')

print('once_v04 ≠ once_v04_taklit:', degisen)
print('notlu satır:', sum(1 for r in S if 'not_taklit' in r), '/', len(NOTLAR))

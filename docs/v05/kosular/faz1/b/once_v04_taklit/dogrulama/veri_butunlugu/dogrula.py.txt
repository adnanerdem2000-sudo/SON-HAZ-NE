def komut_coz(satir):
    s = satir.split('# komut:', 1)[1].strip()
    lx = shlex.shlex(s, posix=True)
    lx.whitespace_split = True
    lx.commenters = ''
    lx.escape = ''
    t = list(lx)
    out = {'_on': [], 'secenek': {}, 'izin_hata': []}
    i = 0
    # önek: tk timeout N node <koşucu> <senaryo>
    while i < len(t) and not t[i].startswith('--'):
        out['_on'].append(t[i]); i += 1
    while i < len(t):
        k = t[i]
        if k == '--izin-hata':
            desen = t[i + 1]; i += 2
            ger = None
            if i < len(t) and t[i] == '#':
                ger = ' '.join(t[i + 1:]); i = len(t)
            elif '#' in desen:
                desen, ger = [x.strip() for x in desen.split('#', 1)]
            out['izin_hata'].append((desen, ger))
            continue
        if i + 1 < len(t) and not t[i + 1].startswith('--'):
            out['secenek'][k] = t[i + 1]; i += 2
        else:
            out['secenek'][k] = True; i += 1
    return out

def on_coz(on):
    # ['/home/user/devtools/tk','timeout','590','node','tests/harness/run.cjs','tests/scenarios/x.js']
    d = {'tk': os.path.basename(on[0]) if on else None}
    if 'timeout' in on:
        d['tk_timeout'] = on[on.index('timeout') + 1]
    if 'node' in on:
        j = on.index('node')
        d['kosucu'] = os.path.basename(on[j + 1]) if j + 1 < len(on) else None
        d['senaryo'] = on[j + 2] if j + 2 < len(on) else None
    return d

def kontroller(r):
    ks = r.get('kontroller') or []
    sozluk, cift = {}, []
    for k in ks:
        if k['id'] in sozluk:
            cift.append(k['id'])
        sozluk.setdefault(k['id'], []).append(k)
    return sozluk, cift

SAYI = re.compile(r'-?\d+(?:[.,]\d+)?(?:e[-+]?\d+)?')

def norm(x):
    return SAYI.sub('#', x if isinstance(x, str) else json.dumps(x, ensure_ascii=False, sort_keys=True))

def tur(v):
    if v is None: return 'null'
    if isinstance(v, bool): return 'bool'
    if isinstance(v, (int, float)): return 'sayi'
    if isinstance(v, str): return 'dizge'
    if isinstance(v, list): return 'dizi'
    if isinstance(v, dict): return 'nesne'
    return type(v).__name__

def anahtar_ara(o, adlar, yol=''):
    bul = []
    if isinstance(o, dict):
        for k, v in o.items():
            if k in adlar:
                bul.append((yol + '.' + k, v))
            if k not in ('steps', 'kontroller', 'checks', 'meta', 'uyarilar'):
                bul += anahtar_ara(v, adlar, yol + '.' + k)
    return bul
